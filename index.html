完整 Google Apps Script 部署設定教學

透過下列完整腳本，即可實現跨裝置 `doPost` 寫入與 `doGet` 讀取串接。

1

建立 Google 試算表

前往 Google Drive 建立試算表，第一頁命名為 候位點餐紀錄，並在第一列設定標題：

ID | 號碼 | 顧客姓名 | 電話 | 人數 | 點餐項目 | 總金額 | 備註 | 狀態 | 時間
2

貼入完整的 Apps Script (Code.gs) 腳本

點選 擴充功能 -> Apps Script，貼入支援雙向讀寫的程式碼：

// Google Apps Script (Code.gs)
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("候位點餐紀錄");
  var data = sheet.getDataRange().getValues();
  var result = [];
  
  for (var i = 1; i < data.length; i++) {
    result.push({
      id: data[i][0],
      ticketNo: data[i][1],
      name: data[i][2],
      phone: data[i][3],
      people: data[i][4],
      items: data[i][5],
      totalPrice: Number(data[i][6]),
      note: data[i][7],
      status: data[i][8],
      time: data[i][9]
    });
  }
  
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("候位點餐紀錄");
    var data = JSON.parse(e.postData.contents);
    
    sheet.appendRow([
      data.id || Date.now(),
      data.ticketNo,
      data.name,
      data.phone,
      data.people,
      data.items,
      data.totalPrice,
      data.note,
      data.status || 'pending',
      data.time || new Date().toLocaleTimeString('zh-TW', { hour: '2-digit', minute: '2-digit' })
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ "result": "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ "result": "error", "message": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
